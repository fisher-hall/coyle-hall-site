While assets/images/ is empty (e.g. a freshly-scaffolded event page awaiting photos) this emits nothing, so there is no empty box or dead prev/next controls. As soon as images are dropped into the folder the carousel appears automatically. Slides are Hugo-processed to responsive WebP, mirroring carousel-slides. Params: dir (image subfolder under assets/images), alt (alt-text prefix), label (aria-label for the carousel region). */ -}} {{- $dir := .Get "dir" -}} {{- $alt := .Get "alt" -}} {{- $label := .Get "label" | default (printf "%s photo carousel" $alt) -}} {{- $imgs := sort (resources.Match (printf "images/%s/*" $dir)) "Name" -}} {{- if gt (len $imgs) 0 -}}
{{- $index := 0 -}} {{- range $imgs -}} {{- $ext := lower (path.Ext .Name) -}} {{- if in (slice ".jpg" ".jpeg" ".png" ".webp" ".gif" ".avif" ".svg") $ext -}} {{- $index = add $index 1 -}} {{- end -}} {{- end -}}
